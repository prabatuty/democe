cmake_minimum_required(VERSION 3.20)
project(mini_mysql LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(APPLE)
  include(CheckIncludeFileCXX)
  check_include_file_cxx(filesystem HAVE_CXX_FILESYSTEM_HEADER)
  if(NOT HAVE_CXX_FILESYSTEM_HEADER)
    execute_process(COMMAND xcrun --show-sdk-path OUTPUT_VARIABLE sdk_root
      OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    if(EXISTS "${sdk_root}/usr/include/c++/v1/filesystem")
      message(STATUS "Using C++ standard library headers from the macOS SDK")
      include_directories(SYSTEM "${sdk_root}/usr/include/c++/v1")
    endif()
  endif()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/MYSQL_VERSION" version_lines)
foreach(line IN LISTS version_lines)
  if(line MATCHES "^([A-Z_]+)=(.*)$")
    set(${CMAKE_MATCH_1} "${CMAKE_MATCH_2}")
  endif()
endforeach()
set(DEMO_EDITION "CE")
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/internal/meb/backup.cxx")
  set(DEMO_EDITION "EE")
endif()
configure_file(src/config.h.in generated/config.h @ONLY)
include(cmake/GenerateErrors.cmake)
add_subdirectory(src)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/internal/meb/backup.cxx")
  add_executable(mysqlbackup internal/meb/backup.cxx)
endif()
enable_testing()
foreach(suite main storage errors)
  add_test(NAME ${suite} COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/mysql-test/run-tests.py
    --source ${CMAKE_CURRENT_SOURCE_DIR} --build ${CMAKE_CURRENT_BINARY_DIR}
    --suite ${suite})
endforeach()
if(TARGET mysqlbackup)
  add_test(NAME meb COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_CURRENT_SOURCE_DIR}/mysql-test/run-tests.py
    --source ${CMAKE_CURRENT_SOURCE_DIR} --build ${CMAKE_CURRENT_BINARY_DIR}
    --suite meb)
endif()
